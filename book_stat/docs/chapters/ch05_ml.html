5.8 ML.08 — Comparison Table

import warnings
warnings.filterwarnings("ignore")
import pandas as pd
from did_multiplegt_stat import DIDMultiplegtStat
from sklearn.ensemble import (RandomForestRegressor, RandomForestClassifier,
                               BaggingRegressor, BaggingClassifier)
from sklearn.neural_network import MLPRegressor, MLPClassifier
from sklearn.tree import DecisionTreeRegressor, DecisionTreeClassifier
from sklearn.linear_model import LinearRegression, LogisticRegression

SEED = 42
gazoline = pd.read_stata(
    "https://github.com/chaisemartinPackages/ApplicationData/raw/main/data_gazoline.dta"
)
import time

strategies = {
    "Default (sklearn)": {},
    "asinstata":         {"asinstata": True},
    "Random Forest":     {
        "model_deltay": RandomForestRegressor(n_estimators=50, max_depth=5, random_state=SEED),
        "model_stayer": RandomForestClassifier(n_estimators=50, max_depth=5, random_state=SEED)},
    "Bagging (trees)":   {
        "model_deltay": BaggingRegressor(estimator=DecisionTreeRegressor(max_depth=4),
                                          n_estimators=50, random_state=SEED),
        "model_stayer": BaggingClassifier(estimator=DecisionTreeClassifier(max_depth=4),
                                           n_estimators=50, random_state=SEED)},
}
for name, kwargs in strategies.items():
    t0 = time.time()
    m = DIDMultiplegtStat(estimator=["waoss"], order=1, controls=["lngpinc"],
                           placebo=1, **kwargs)
    m.fit(gazoline, Y="lngca", ID="id", Time="year", D="tau")
    elapsed = time.time() - t0
    tbl = m.results_["results"]["table"]
    coef = float(tbl.loc["WAS"]["Estimate"])
    se   = float(tbl.loc["WAS"]["SE"])
    print(f"{name:25s}  WAS={coef:+.7f}  SE={se:.7f}  ({elapsed:.1f}s)")
