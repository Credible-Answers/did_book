    "https://github.com/chaisemartinPackages/ApplicationData/raw/main/data_gazoline.dta"
)

# V.04 — reduced form on placebo sample
m = DIDMultiplegtStat(estimator=["waoss"], order=1, controls=["lngpinc"],
